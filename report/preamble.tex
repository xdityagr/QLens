% ---- screenshots ---------------------------------------------------------------------------------
% \screenshot{file}{caption}{label}: includes screenshots/<file>.png when it exists; otherwise draws a
% framed placeholder of the same footprint so the layout (and the list of figures) is final already.
\newcommand{\screenshot}[3]{%
  \begin{figure}[htbp]
    \centering
    \IfFileExists{screenshots/#1.png}{%
      \fcolorbox{rule}{white}{\includegraphics[width=0.96\linewidth,height=0.40\textheight,keepaspectratio]{screenshots/#1.png}}%
    }{%
      \begin{tikzpicture}
        \fill[paper,rounded corners=4pt] (0,0) rectangle (0.96\linewidth,0.50\linewidth);
        \draw[rule,line width=0.8pt,dash pattern=on 4pt off 3pt,rounded corners=4pt]
          (0,0) rectangle (0.96\linewidth,0.50\linewidth);
        \fill[rule!60] (0,0.50\linewidth) ++(0,-14pt) rectangle (0.96\linewidth,0.50\linewidth);
        \foreach \i/\c in {0/lane7,1/lane6,2/lane4} \fill[\c!70] (10pt+\i*9pt,0.50\linewidth-7pt) circle (2.6pt);
        \node[font=\sffamily\small\color{muted},align=center,text width=0.7\linewidth]
          at (0.48\linewidth,0.24\linewidth)
          {\faImage[regular]\par\smallskip Screenshot to be captured from the running application\par
           \texttt{\footnotesize screenshots/\detokenize{#1}.png}};
      \end{tikzpicture}%
    }
    \caption{#2}\label{#3}
  \end{figure}}
