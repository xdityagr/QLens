% Simple three-tier architecture of QLens.
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[
    tier/.style={draw=#1!70!black, fill=#1!6, line width=0.8pt, rounded corners=5pt,
      minimum width=12.4cm, align=center, inner sep=6pt, font=\sffamily\small},
    part/.style={draw=#1!70!black, fill=white, line width=0.6pt, rounded corners=3pt,
      minimum width=5.6cm, minimum height=1.35cm, align=center, font=\sffamily\small},
    arr/.style={-{Stealth[length=6pt,width=5pt]}, line width=0.9pt, draw=ink!80},
    lab/.style={font=\sffamily\footnotesize, align=left, text=muted},
  ]
  \node[tier=lane0, minimum height=1.6cm] (ui) at (0,0)
    {\textbf{1. Browser (user interface)}\\[2pt]
     React web pages: Import, Analyze, Design, Data, Ask AI, History};
  \node[tier=lane1, minimum height=1.6cm] (app) at (0,-3.0)
    {\textbf{2. Application server (FastAPI, Python)}\\[2pt]
     receives requests, runs the analysis, builds the design, sends SQL to the database};
  \node[tier=lane4, minimum height=3.4cm] (db) at (0,-7.7) {};
  \node[font=\sffamily\small\bfseries, anchor=north] at (db.north) {\strut 3. Database server (PostgreSQL 18)};
  \node[part=lane1] (meta) at (-3.05,-8.05) {\textbf{schema \code{qlens}}\\ QLens's own data:\\ users, datasets, designs, runs};
  \node[part=lane3] (ws) at (3.05,-8.05) {\textbf{one schema per workspace}\\ the uploaded sheet and the\\ generated tables};
  \draw[arr, {Stealth[length=6pt,width=5pt]}-{Stealth[length=6pt,width=5pt]}] (ui) -- (app)
    node[lab, midway, right=3pt] {HTTP requests with JSON data};
  \draw[arr] ([xshift=-3.05cm]app.south) -- (meta.north)
    node[lab, midway, left=3pt, align=right] {SQL as the owner\\ user \code{qlens\_app}};
  \draw[arr] ([xshift=3.05cm]app.south) -- (ws.north)
    node[lab, midway, right=3pt] {user SQL as a restricted\\ user \code{qlens\_ws\_<id>}};
\end{tikzpicture}
\caption[System architecture]{Three-tier architecture of QLens. The application talks to PostgreSQL as
two different database users: the owner for its own work, and a restricted read-only user (one per
workspace) for the queries that people type.}
\label{fig:architecture}
\end{figure}
