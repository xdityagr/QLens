% Worked example: the fee register's columns with the dependencies QLens discovers (dependency diagram).

\begin{figure}[tbp]
\centering
\begin{tikzpicture}[
    col/.style={draw=rule, fill=paper, font=\ttfamily\scriptsize, minimum height=6mm, inner xsep=2pt},
    dep/.style={-{Stealth[length=4pt,width=3.4pt]}, line width=0.7pt, draw=#1},
  ]
  \node[col, minimum width=2.05cm] (sn) at (1.0,0) {student\_name};
  \node[col, minimum width=2.15cm, right=0pt of sn] (sp) {student\_phone};
  \node[col, minimum width=1.25cm, right=0pt of sp] (b) {batch};
  \node[col, minimum width=1.5cm, right=0pt of b] (t) {teacher};
  \node[col, minimum width=2.15cm, right=0pt of t] (tp) {teacher\_phone};
  \node[col, minimum width=1.2cm, right=0pt of tp] (m) {month};
  \node[col, minimum width=1.6cm, right=0pt of m] (f) {fee\_paid};
  \node[col, minimum width=2.1cm, right=0pt of f] (pm) {payment\_mode};
  % arcs (colour = table that will hold the dependency)
  \draw[dep=lane0] ([xshift=-3mm]sn.north) to[out=70,in=110] node[above, font=\tiny, text=lane0] {$\leftrightarrow$} ([xshift=-3mm]sp.north);
  \draw[dep=lane0] ([xshift=3mm]sp.north) to[out=70,in=110] ([xshift=-2mm]b.north);
  \draw[dep=lane1] ([xshift=2mm]b.north) to[out=70,in=110] ([xshift=-3mm]t.north);
  \draw[dep=lane2] ([xshift=3mm]t.north) to[out=70,in=110] node[above, font=\tiny, text=lane2] {$\leftrightarrow$} ([xshift=-3mm]tp.north);
  % composite determinant (student_name, month) -> fee_paid, payment_mode, drawn below
  \draw[lane3, line width=0.7pt] (sn.south) -- ++(0,-0.45) coordinate (k1) -- (k1 -| m.south) -- (m.south);
  \draw[dep=lane3] (k1 -| m.south) ++(0,0) -- ++(0.9,0) -| (f.south);
  \draw[dep=lane3] (k1 -| f.south) -- ++(1.0,0) -| (pm.south);
  \node[font=\sffamily\tiny, text=lane3, anchor=north west] at ([yshift=-1pt]k1) {key (student\_name, month) determines fee\_paid and payment\_mode};
\end{tikzpicture}
\caption[Fee-register dependency diagram]{Dependency diagram of the fee register: arrows point from

and \fd{teacher}{teacher\_phone} on 5.}

\end{figure}
