% Normalization pipeline from upload to the lossless-join check (simple flowchart).
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[
    step/.style={draw=#1!75!black, fill=#1!8, line width=0.6pt, rounded corners=3pt, align=left,
      text width=10.2cm, inner sep=4pt, font=\sffamily\footnotesize},
    dec/.style={diamond, aspect=2.6, draw=lane7!80!black, fill=lane7!6, line width=0.6pt,
      font=\sffamily\footnotesize, align=center, inner sep=1pt},
    num/.style={circle, fill=#1, text=white, font=\sffamily\scriptsize\bfseries, inner sep=1pt, minimum size=4.5mm},
    flow/.style={-{Stealth[length=5pt,width=4pt]}, line width=0.7pt, draw=ink!80},
  ]
  \node[step=lane0] (s1) at (0,0) {\textbf{Import.} Copy the sheet into a raw table (\code{COPY}), decide each column's type with SQL, build a typed copy.};
  \node[step=lane0, below=3.5mm of s1] (s2) {\textbf{Profile.} Count empty and distinct values of every column with aggregate queries.};
  \node[step=lane4, below=3.5mm of s2] (s3) {\textbf{Find keys and functional dependencies.} Stored functions \code{is\_unique} and \code{check\_fd} test them with \code{GROUP BY}.};
  \node[step=lane2, below=3.5mm of s3] (s4) {\textbf{Clean the data.} Merge similar spellings; give each determinant one value (the user decides). Every change is logged.};
  \node[step=lane1, below=3.5mm of s4] (s5) {\textbf{Design.} Canonical cover $\rightarrow$ 3NF decomposition (\cref{alg:synthesis}) $\rightarrow$ tables with primary and foreign keys.};
  \node[step=lane3, below=3.5mm of s5] (s6) {\textbf{Apply in one transaction.} \code{CREATE TABLE}s, fill them with \code{INSERT \dots{} SELECT}, then run the lossless-join check.};
  \node[dec, below=5mm of s6] (d) {all rows\\come back?};
  \node[step=lane4, below left=6mm and -2.2cm of d, text width=4.2cm] (ok) {\textbf{yes: COMMIT.} The new tables are kept.};
  \node[step=lane7, below right=6mm and -2.2cm of d, text width=4.2cm] (bad) {\textbf{no: ROLLBACK.} Nothing changes.};
  \foreach \a/\b in {s1/s2,s2/s3,s3/s4,s4/s5,s5/s6,s6/d} \draw[flow] (\a) -- (\b);
  \draw[flow] (d.south west) -- (ok.north);
  \draw[flow] (d.south east) -- (bad.north);
  \draw[flow] (s4.east) -- ++(0.45,0) |- (s3.east) node[pos=0.25, right, font=\sffamily\scriptsize] {re-check};
  \foreach \s/\c [count=\i] in {s1/lane0,s2/lane0,s3/lane4,s4/lane2,s5/lane1,s6/lane3}
    \node[num=\c] at (\s.west) {\i};
\end{tikzpicture}
\caption[From sheet to normalized tables]{The steps from an uploaded sheet to normalized tables. Steps 3 and
4 repeat until the user is happy with the data; step 6 keeps the new tables only if the check passes.}
\label{fig:pipeline}
\end{figure}
