% End-to-end normalization pipeline: what runs where, and which tables each step reads and writes.
\begin{figure}[tbp]
\centering
\begin{tikzpicture}[
    step/.style={draw=#1!75!black, fill=#1!8, line width=0.6pt, rounded corners=3pt, align=left,
      text width=6.6cm, inner sep=4pt, font=\sffamily\scriptsize},
    step/.default=lane0,
    store/.style={draw=rule, fill=paper, rounded corners=2pt, align=left, text width=4.6cm,
      inner sep=3pt, font=\sffamily\scriptsize\color{ink}},
    dec/.style={diamond, aspect=2.4, draw=lane7!80!black, fill=lane7!6, line width=0.6pt,
      font=\sffamily\scriptsize, align=center, inner sep=1pt},
    num/.style={circle, fill=#1, text=white, font=\sffamily\scriptsize\bfseries, inner sep=1.3pt, minimum size=4.2mm},
    flow/.style={-{Stealth[length=4.5pt,width=3.6pt]}, line width=0.6pt, draw=ink!80},
    link/.style={line width=0.4pt, draw=muted, dash pattern=on 1.5pt off 1.5pt},
  ]
  \node[step=lane0] (s1) at (0,0) {\textbf{Upload and parse.} CSV (encoding and delimiter sniffed) or the first XLSX sheet; headers become safe identifiers. Limits: 20\,MB, 100{,}000 rows, 60 columns.};
  \node[step=lane0, below=4mm of s1] (s2) {\textbf{Stage.} \code{COPY} every cell as text into \code{\_raw}; infer each column's type in SQL (regex tests, \code{qlens.try\_date}); \code{CREATE TABLE \_staging AS SELECT} with casts.};
  \node[step=lane0, below=4mm of s2] (s3) {\textbf{Profile.} One aggregate pass for null and distinct counts; most frequent sample values per column.};
  \node[step=lane4, below=4mm of s3] (s4) {\textbf{Discover keys and dependencies in SQL.} \code{qlens.is\_unique} on 1--3 column combinations; \code{qlens.check\_fd} for every eligible LHS of size 1--2; exact ($0$ violating rows) and approximate ($\le 2\%$) dependencies kept; derived ones flagged.};
  \node[step=lane2, below=4mm of s4] (s5) {\textbf{Find problems.} Approximate dependencies become inconsistencies with their conflicting groups; near-duplicate spellings via normalized form and \code{pg\_trgm} similarity over a GIN index.};
  \node[step=lane2, below=4mm of s5] (s6) {\textbf{Clean up} (user decides). Merge duplicates, accept the suggested value per conflict, or ``accept all''. Each edit is an \code{UPDATE} of \code{\_staging} in a transaction, logged and followed by re-profiling and re-analysis.};
  \node[step=lane1, below=4mm of s6] (s7) {\textbf{Synthesize 3NF.} Minimal cover of the exact, accepted dependencies; Bernstein synthesis with key relation (\cref{alg:bernstein}); names, surrogate keys, natural \code{UNIQUE} keys and foreign keys; DDL generated.};
  \node[step=lane3, below=4mm of s7] (s8) {\textbf{Apply in one transaction.} \code{CREATE TABLE}s in dependency order, \code{INSERT \dots{} SELECT} from \code{\_staging} with natural-key lookups, view \code{original\_sheet}.};
  \node[dec, below=5mm of s8] (d) {lossless?\\\code{EXCEPT ALL} both\\ways $= 0$};
  \node[step=lane4, below left=6mm and -18mm of d, text width=3.4cm] (ok) {\textbf{COMMIT}; proposal \code{applied}, dataset \code{applied}.};
  \node[step=lane7, below right=6mm and -18mm of d, text width=3.4cm] (bad) {\textbf{ROLLBACK}; previous schema untouched, mismatching rows shown.};

  \foreach \a/\b in {s1/s2,s2/s3,s3/s4,s4/s5,s5/s6,s6/s7,s7/s8,s8/d} \draw[flow] (\a) -- (\b);
  \draw[flow] (d.south west) ++(0.2,0.05) -- (ok.north) node[midway, left, font=\sffamily\scriptsize] {yes};
  \draw[flow] (d.south east) ++(-0.2,0.05) -- (bad.north) node[midway, right, font=\sffamily\scriptsize] {no};
  \draw[flow] (s6.west) -- ++(-0.45,0) |- (s4.west) node[pos=0.25, left, font=\sffamily\scriptsize, align=right] {re-\\analyze};

  % numbered badges
  \foreach \s/\c [count=\i] in {s1/lane0,s2/lane0,s3/lane0,s4/lane4,s5/lane2,s6/lane2,s7/lane1,s8/lane3}
    \node[num=\c] at (\s.north west) {\i};

  % data touched by each step
  \node[store, right=9mm of s2] (t2) {\code{ws\_<id>\_src.\_raw} (text, never edited)\\ \code{ws\_<id>\_src.\_staging} (typed)};
  \node[store, right=9mm of s3] (t3) {\code{qlens.datasets}, \code{qlens.dataset\_columns}};
  \node[store, right=9mm of s4] (t4) {\code{qlens.candidate\_keys} (+ columns)\\ \code{qlens.functional\_dependencies} (+ LHS)};
  \node[store, right=9mm of s5] (t5) {\code{qlens.duplicate\_groups} / \code{\_values}};
  \node[store, right=9mm of s6] (t6) {\code{\_staging} rows \textrightarrow{} \code{qlens.dataset\_fixes}};
  \node[store, right=9mm of s7] (t7) {\code{qlens.schema\_proposals}, \code{proposed\_tables}, \code{proposed\_columns}, \code{proposed\_unique\_keys}};
  \node[store, right=9mm of s8] (t8) {\code{ws\_<id>}: generated tables and view \code{original\_sheet}};
  \foreach \i in {2,...,8} \draw[link] (s\i.east) -- (t\i.west);

  \node[font=\sffamily\scriptsize\bfseries, color=muted, anchor=south] at (s1.north) {\strut work done by the application and the database};
  \node[font=\sffamily\scriptsize\bfseries, color=muted, anchor=south] at (t2.north |- s1.north) {\strut tables written};
  % legend for colours
  \node[font=\sffamily\scriptsize, align=left, anchor=north west, text width=4.6cm] at (t8.south west |- d.north)
    {\textcolor{lane0}{\rule{6pt}{6pt}} import \enspace \textcolor{lane4}{\rule{6pt}{6pt}} discovery\\
     \textcolor{lane2}{\rule{6pt}{6pt}} cleaning \enspace \textcolor{lane1}{\rule{6pt}{6pt}} design \enspace
     \textcolor{lane3}{\rule{6pt}{6pt}} apply\\[2pt]
     \color{muted} Steps 1--7 are built and tested; the apply transaction (8) runs the generated SQL that the
     end-to-end tests already execute and verify.};
\end{tikzpicture}
\caption{The normalization pipeline from upload to the lossless check. Left: the steps in order (colour
marks the phase); right: the tables each step writes. Steps 4--6 can loop: every clean-up re-profiles the
edited columns and re-runs discovery.}\label{fig:pipeline}
\end{figure}
