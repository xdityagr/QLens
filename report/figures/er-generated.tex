\centering
\resizebox{\linewidth}{!}{%
\begin{tikzpicture}[
    ent/.style={rectangle split, rectangle split parts=2, draw=#1!75!black, line width=0.7pt,

      text width=2.75cm},
    rel/.style={diamond, aspect=2.0, draw=ink!70, fill=white, line width=0.6pt, inner sep=1pt,
      font=\sffamily\scriptsize, align=center},
    ln/.style={line width=0.6pt, draw=ink!75},
    card/.style={font=\sffamily\footnotesize\bfseries, text=lane7!80!black, fill=white, inner sep=1pt},
  ]
  \node[ent=lane3] (F) at (0,0) {\centering\textbf{Fee Record}\nodepart{two}\underline{id}\\ month\\ fee\_paid\\ payment\_mode};
  \node[ent=lane0] (S) at (4.6,0) {\centering\textbf{Student}\nodepart{two}\underline{id}\\ student\_name\\ student\_phone};
  \node[ent=lane1] (B) at (9.2,0) {\centering\textbf{Batch}\nodepart{two}\underline{id}\\ batch};
  \node[ent=lane2] (T) at (13.8,0) {\centering\textbf{Teacher}\nodepart{two}\underline{id}\\ teacher\\ teacher\_phone};
  \node[rel] (r1) at (2.3,-1.6) {paid by};
  \node[rel] (r2) at (6.9,-1.6) {enrolled\\in};
  \node[rel] (r3) at (11.5,-1.6) {taught\\by};
  \draw[ln] (F.south) |- (r1.west) node[card, pos=0.3, left] {N};
  \draw[ln] (r1.east) -| (S.south) node[card, pos=0.75, right] {1};
  \draw[ln] (S.south east) ++(-0.3,0) |- (r2.west) node[card, pos=0.3, right] {N};
  \draw[ln] (r2.east) -| (B.south) node[card, pos=0.75, right] {1};
  \draw[ln] (B.south east) ++(-0.3,0) |- (r3.west) node[card, pos=0.3, right] {N};
  \draw[ln] (r3.east) -| (T.south) node[card, pos=0.75, right] {1};
  \foreach \n/\r in {F/{3{,}000 rows}, S/{300 rows}, B/{8 rows}, T/{6 rows}}
    \node[font=\sffamily\scriptsize\color{muted}, anchor=south] at (\n.north) {\r};
\end{tikzpicture}}
\caption[ER diagram of the generated fee-register database]{ER diagram of the database QLens generates
for the fee register. Each fee record is paid by one student, each student is enrolled in one batch and
each batch is taught by one teacher (all N\,:\,1). Converted to tables, each N side gets a foreign key
(\code{student\_id}, \code{batch\_id}, \code{teacher\_id}).}
\label{fig:er-generated}
\end{figure}
