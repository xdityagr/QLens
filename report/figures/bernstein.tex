% 3NF decomposition (synthesis) algorithm as used by QLens (textbook form, Korth 7e §7.5).
\begin{algorithm}[htbp]
\caption[3NF decomposition algorithm]{3NF decomposition (synthesis) using a canonical cover, as used by QLens}\label{alg:synthesis}
\KwIn{relation schema $R$ (the sheet's columns); set $F$ of functional dependencies}
\KwOut{relation schemas $R_1,\dots,R_n$ in 3NF; the decomposition is lossless and dependency-preserving}
\BlankLine
$F_c \leftarrow$ a canonical cover of $F$\;
$i \leftarrow 0$\;
\ForEach{dependency $\alpha \rightarrow \beta$ in $F_c$}{
  $i \leftarrow i+1$;\quad $R_i \leftarrow \alpha\beta$\tcp*[r]{one schema per dependency}
}
\If{no $R_j$ contains a candidate key of $R$}{
  $i \leftarrow i+1$;\quad $R_i \leftarrow$ a candidate key of $R$\tcp*[r]{guarantees a lossless join}
}
\Repeat{no more changes}{
  \lIf{some $R_j \subseteq R_k$ ($j \neq k$)}{delete $R_j$\tcp*[r]{remove redundant schemas}}
}
\Return{$R_1,\dots,R_i$}\;
\end{algorithm}
