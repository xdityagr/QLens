% 3NF synthesis as implemented in backend/qlens/analysis/fd_algebra.py (synthesize_3nf).
\begin{algorithm}[htbp]
\caption[Lossless 3NF synthesis]{Lossless 3NF synthesis used by QLens (Bernstein synthesis with equivalence merging, the
step-5 redundancy pruning and a key relation)}\label{alg:bernstein}
\KwIn{attributes $U$ in sheet order; exact, accepted dependencies $F$; candidate keys $K_1,\dots,K_m$ found in the data}
\KwOut{relations $R_1,\dots,R_n$ (entities) and one fact relation $R_f$; every $R_i$ is in 3NF and $\bowtie_i R_i$ is lossless}
\BlankLine
$G \leftarrow \textsc{MinimalCover}(F)$ \tcp*{singleton RHS, left-reduced, no redundant FD}
partition $G$ into groups $G_X=\{X\rightarrow A \in G\}$, one per left-hand side $X$\;
\ForEach{pair of groups $G_X, G_Y$}{
  \If(\tcp*[f]{equivalent determinants}){$X \subseteq Y^{+}_{G}$ \textbf{and} $Y \subseteq X^{+}_{G}$}{merge $G_X$ and $G_Y$ into one group with determinants $\{X,Y\}$\;}
}
$J \leftarrow \{X\rightarrow B \mid X, Y \text{ determinants of one merged group}, B\in Y\}$\;
\For(\tcp*[f]{Bernstein step 5}){$X\rightarrow A \in G$ \textbf{from last to first}}{
  \lIf{$A \in X^{+}_{(G\setminus\{X\rightarrow A\})\cup J}$}{remove $X\rightarrow A$ from $G$}
}
\ForEach{merged group with determinants $X_1,\dots,X_k$}{
  $R \leftarrow X_1\cup\dots\cup X_k \cup \{A \mid X_j\rightarrow A \in G\}$; \quad key $\leftarrow$ earliest $X_j$ in sheet order\;
  emit entity relation $R$ unless it carries no dependency\;
}
$D \leftarrow$ attributes appearing on the right of some emitted relation's non-key part\;
\eIf{some emitted $R$ contains $U\setminus D$ \textbf{and} $R^{+}_{G}=U$}{mark $R$ as the fact relation $R_f$\;}{
  $Z \leftarrow U$\;
  \For(\tcp*[f]{FD-key by reverse reduction}){$a \in U$ \textbf{in reverse sheet order}}{
    \lIf{$(Z\setminus\{a\})^{+}_{G} = U$}{$Z \leftarrow Z\setminus\{a\}$}
  }
  $R_f \leftarrow Z$; its unique key is $K_1$ if $K_1\subseteq Z$, else none (surrogate key only)\;
}
drop every entity relation whose attributes are a subset of another relation's\;
\Return{entity relations, then $R_f$}\;
\end{algorithm}
